object ShizukuHelper {

    private const val REQ = 1001

    fun hasPermission() = runCatching {
        Shizuku.pingBinder() &&
                Shizuku.checkSelfPermission() == PackageManager.PERMISSION_GRANTED
    }.getOrDefault(false)

    fun requestPermission(context: Context) {
        try {
            if (!Shizuku.pingBinder()) {
                // Shizuku is not running
                Toast.makeText(context, "Shizuku is not running", Toast.LENGTH_SHORT).show()
                return
            }

            when (Shizuku.checkSelfPermission()) {
                PackageManager.PERMISSION_GRANTED -> {
                    Toast.makeText(context, "Permission already granted", Toast.LENGTH_SHORT).show()
                }
                PackageManager.PERMISSION_DENIED -> {
                    Shizuku.requestPermission(REQ)
                }
            }
        } catch (e: Throwable) {
            // Shizuku not installed or other binder error
            Toast.makeText(context, "Shizuku not installed or unavailable", Toast.LENGTH_SHORT).show()
        }
    }

}

    val rideRequestsList: MutableState<List<RideRequest>> =  mutableStateOf(emptyList())

    @SuppressLint("CoroutineCreationDuringComposition")
    @Composable
    fun DriverPROApp() {
        val context = LocalContext.current
        val token = getToken(context, "JWT_TOKEN")

        // Create a mutable state for the user
        val user = remember { mutableStateOf<User?>(null) }

        // clearAllRequests(context, "RIDE-REQUESTS")
//        clearAllRequests(context, "RIDE-REQUESTS-TEMP")


        // Call the API to get user info once the token is retrieved
        LaunchedEffect(token) {
            if (token != null) {
                // Call the getUser function asynchronously and update the user state
                val result = getUser(token)
                if (result.isSuccess) {
                    user.value = result.getOrNull()  // Update user info if successful
                } else {
                    // Handle failure (e.g., show an error message)
                    Log.e("DriverPROApp", "Failed to fetch user info: ${result.exceptionOrNull()?.message}")
                }
            }
        }

        // Fetch the ride requests if the user is subscribed
        val rideRequests = getRideRequestArray(context, "RIDE-REQUESTS")



        // Update the ride requests list if the user is subscribed
        LaunchedEffect(user.value) {
            if (user.value?.subscribed == true) {
                rideRequestsList.value = rideRequests?.toList() ?: emptyList()
            }
        }

        // Check if the user is authenticated and proceed accordingly
        if (user.value != null) {
            // If the user is authenticated, show the main ride screen
//            MainRideScreen(user = user.value!!, rideRequests = rideRequestsList.value)
            MainScreen(user = user.value!!, rideRequests = rideRequestsList.value)
        } else {
            // If no user found, show the login screen
            LoginScreen { loggedInUser ->
                user.value = loggedInUser  // Set the logged-in user
            }
        }
    }

    private fun startScreenCapture() {
        val intent = mediaProjectionManager.createScreenCaptureIntent()
        startActivityForResult(intent, REQUEST_CODE_CAPTURE)
    }

    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)

        if (requestCode == REQUEST_CODE_CAPTURE && resultCode == RESULT_OK && data != null) {

            val serviceIntent = Intent(this, ScreenCaptureService::class.java).apply {
                putExtra("code", resultCode)
                putExtra("data", data)
            }

            Log.d("service data", data.toString())

            // IMPORTANT → MUST BE FOREGROUND SERVICE
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                Log.d("service data", "forground")
                startForegroundService(serviceIntent)
            } else {
                Log.d("service data", "background")
                startService(serviceIntent)
            }

        }
    }


    override fun onResume() {
